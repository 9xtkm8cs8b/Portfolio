        text-align: center;

        background: #0b1120;

        border: 2px solid #94a3b8;

        border-radius: 8px;

        box-shadow: 0 0 20px rgba(148, 163, 184, 0.2);

        max-width: 95vw;

      }

        margin-top: 0.75rem;

        font-size: 0.9rem;

  <body>

    <main class="game-shell">

      <h1>Mini Pong</h1>

      <p id="scoreboard">0 : 0</p>

      <canvas id="game" width="800" height="450" aria-label="Pong game board"></canvas>

      <div class="hint">Left: W/S &nbsp;|&nbsp; Right: ↑/↓ &nbsp;|&nbsp; Space: restart</div>

    </main>


    <script>

      const canvas = document.getElementById('game');

      const ctx = canvas.getContext('2d');

      const scoreboard = document.getElementById('scoreboard');


      const paddleWidth = 14;

      const paddleHeight = 95;

      const paddleSpeed = 7;

      const ballRadius = 10;


      const leftPaddle = {

        x: 28,

        y: canvas.height / 2 - paddleHeight / 2,

        score: 0,

        up: 'KeyW',

        down: 'KeyS'

      };


      const rightPaddle = {

        x: canvas.width - 28 - paddleWidth,

        y: canvas.height / 2 - paddleHeight / 2,

        score: 0,

        up: 'ArrowUp',

        down: 'ArrowDown'

      };


      const ball = {

        x: canvas.width / 2,

        y: canvas.height / 2,

        vx: 5,

        vy: 3

      };


      const pressed = new Set();


      function resetBall(direction = 1) {

        ball.x = canvas.width / 2;

        ball.y = canvas.height / 2;

        const horizontal = 4 + Math.random() * 2;

        const vertical = (Math.random() * 4 - 2) || 1;

        ball.vx = horizontal * direction;

        ball.vy = vertical;

      }


      function clampPaddle(paddle) {

        paddle.y = Math.max(0, Math.min(canvas.height - paddleHeight, paddle.y));

      }


      function updatePaddles() {

        if (pressed.has(leftPaddle.up)) leftPaddle.y -= paddleSpeed;

        if (pressed.has(leftPaddle.down)) leftPaddle.y += paddleSpeed;

        if (pressed.has(rightPaddle.up)) rightPaddle.y -= paddleSpeed;

        if (pressed.has(rightPaddle.down)) rightPaddle.y += paddleSpeed;


        clampPaddle(leftPaddle);

        clampPaddle(rightPaddle);

      }


      function ballHitsPaddle(paddle) {

        return (

          ball.x - ballRadius < paddle.x + paddleWidth &&

          ball.x + ballRadius > paddle.x &&

          ball.y + ballRadius > paddle.y &&

          ball.y - ballRadius < paddle.y + paddleHeight

        );

      }


      function updateBall() {

        ball.x += ball.vx;

        ball.y += ball.vy;


        if (ball.y - ballRadius <= 0 || ball.y + ballRadius >= canvas.height) {

          ball.vy *= -1;

        }


        if (ballHitsPaddle(leftPaddle) && ball.vx < 0) {

          ball.x = leftPaddle.x + paddleWidth + ballRadius;

          ball.vx *= -1.05;

          ball.vy += (ball.y - (leftPaddle.y + paddleHeight / 2)) * 0.04;

        }


        if (ballHitsPaddle(rightPaddle) && ball.vx > 0) {

          ball.x = rightPaddle.x - ballRadius;

          ball.vx *= -1.05;

          ball.vy += (ball.y - (rightPaddle.y + paddleHeight / 2)) * 0.04;

        }


        if (ball.x + ballRadius < 0) {

          rightPaddle.score += 1;

          resetBall(-1);

        }


        if (ball.x - ballRadius > canvas.width) {

          leftPaddle.score += 1;

          resetBall(1);

        }

      }


      function drawNet() {

        ctx.fillStyle = '#334155';

        const dashHeight = 18;

        for (let y = 0; y < canvas.height; y += dashHeight * 2) {

          ctx.fillRect(canvas.width / 2 - 2, y, 4, dashHeight);

        }

      }


      function drawPaddle(paddle) {

        ctx.fillStyle = '#f8fafc';

        ctx.fillRect(paddle.x, paddle.y, paddleWidth, paddleHeight);

      }


      function drawBall() {

        ctx.fillStyle = '#38bdf8';

        ctx.beginPath();

        ctx.arc(ball.x, ball.y, ballRadius, 0, Math.PI * 2);

        ctx.fill();

      }


      function render() {

        ctx.clearRect(0, 0, canvas.width, canvas.height);

        drawNet();

        drawPaddle(leftPaddle);

        drawPaddle(rightPaddle);

        drawBall();

        scoreboard.textContent = `${leftPaddle.score} : ${rightPaddle.score}`;

      }


      function gameLoop() {

        updatePaddles();

        updateBall();

        render();

        requestAnimationFrame(gameLoop);

      }


      window.addEventListener('keydown', (event) => {

        pressed.add(event.code);

        if (event.code === 'Space') {

          leftPaddle.score = 0;

          rightPaddle.score = 0;

          resetBall(Math.random() < 0.5 ? -1 : 1);

        }

      });


      window.addEventListener('keyup', (event) => {

        pressed.delete(event.code);

      });


      resetBall(Math.random() < 0.5 ? -1 : 1);

      gameLoop();

    </script>

  </body>
